cmake_minimum_required(VERSION 3.22)
project(EmuCoreNES LANGUAGES C CXX)
file(GLOB_RECURSE nestopia_sources CONFIGURE_DEPENDS nestopia/src/*.cpp)
add_library(emucored_nestopia STATIC ${nestopia_sources})
set_target_properties(emucored_nestopia PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_compile_features(emucored_nestopia PUBLIC cxx_std_17)
target_compile_definitions(emucored_nestopia PUBLIC NST_NO_ZLIB NDEBUG)
target_include_directories(emucored_nestopia PUBLIC nestopia/src)
option(EMUCORED_BUILD_CORE_TESTS "Build standalone native session contracts" OFF)
option(EMUCORED_BUILD_ANDROID_RUNTIME "Build own Android runtime" OFF)
if(ANDROID AND EMUCORED_BUILD_ANDROID_RUNTIME)
 add_library(emucored_nes SHARED bridge/session.cpp)
 target_include_directories(emucored_nes PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core-common")
 target_link_libraries(emucored_nes PRIVATE emucored_nestopia android aaudio EGL GLESv2)

 if(EMUCORED_BUILD_CORE_TESTS)
  add_executable(emucored_nes_session_contract bridge/session_contract.cpp)
  target_include_directories(emucored_nes_session_contract PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../core-common")
  target_compile_features(emucored_nes_session_contract PRIVATE cxx_std_17)
  target_link_libraries(emucored_nes_session_contract PRIVATE emucored_nestopia android aaudio EGL GLESv2)
 endif()
endif()
